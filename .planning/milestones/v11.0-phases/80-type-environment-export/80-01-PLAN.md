```fsharp
module FunLang.Tests.TypeEnvTests

open Expecto
open FunLang.IndentFilter

// Reuse the same lex/parse helpers as TypeAnnotationTests
let lexAndFilter (input: string) =
    let lexbuf = FSharp.Text.Lexing.LexBuffer<_>.FromString input
    Lexer.setInitialPos lexbuf "test"
    let rec collect () =
        let tok = Lexer.tokenize lexbuf
        if tok = Parser.EOF then [Parser.EOF]
        else tok :: collect ()
    let rawTokens = collect ()
    filter defaultConfig rawTokens |> Seq.toList

let parseModule (input: string) : Ast.Module =
    let lexbuf = FSharp.Text.Lexing.LexBuffer<_>.FromString input
    Lexer.setInitialPos lexbuf "test"
    let filteredTokens = lexAndFilter input
    let mutable index = 0
    let tokenizer (lexbuf: FSharp.Text.Lexing.LexBuffer<_>) =
        if index < filteredTokens.Length then
            let tok = filteredTokens.[index]
            index <- index + 1
            tok
        else
            Parser.EOF
    Parser.parseModule tokenizer lexbuf

/// Type-check and return the TypeEnv (4th element of Ok tuple from typeCheckModule).
/// typeCheckModule passes empty prelude to typeCheckModuleWithPrelude,
/// so the returned env = initialTypeEnv (builtins) + user bindings.
let typeCheckGetEnv (input: string) : Type.TypeEnv =
    let m = parseModule input
    match TypeCheck.typeCheckModule m with
    | Error errs -> failwith (sprintf "Type check failed: %A" errs)
    | Ok (_warnings, _recEnv, _modules, typeEnv) -> typeEnv

[<Tests>]
let typeEnvTests =
    testList "TypeEnv" [
        // TE-01: user-defined top-level let bindings are in the env
        test "TE-01: let x = 42 produces x : int in env" {
            let env = typeCheckGetEnv "let x = 42"
            match Map.tryFind "x" env with
            | Some (Type.Scheme([], [], Type.TInt)) -> ()
            | Some other -> failwith (sprintf "Expected Scheme([],[],TInt), got %A" other)
            | None -> failwith "x not found in binding env"
        }

        test "TE-01: let f x = x + 1 produces f : int -> int in env" {
            let env = typeCheckGetEnv "let f x = x + 1"
            match Map.tryFind "f" env with
            | Some (Type.Scheme([], [], Type.TArrow(Type.TInt, Type.TInt))) -> ()
            | Some other -> failwith (sprintf "Expected int->int scheme, got %A" other)
            | None -> failwith "f not found in binding env"
        }

        test "TE-01: let id x = x produces polymorphic id in env" {
            let env = typeCheckGetEnv "let id x = x"
            match Map.tryFind "id" env with
            | Some (Type.Scheme([_], [], Type.TArrow(Type.TVar _, Type.TVar _))) -> ()
            | Some other -> failwith (sprintf "Expected polymorphic a->a scheme, got %A" other)
            | None -> failwith "id not found in binding env"
        }

        // TE-02: builtin functions are in the env
        test "TE-02: builtin print is in env" {
            let env = typeCheckGetEnv "let x = 1"
            Expect.isSome (Map.tryFind "print" env) "builtin 'print' should be in env"
        }

        test "TE-02: builtin string_length is in env" {
            let env = typeCheckGetEnv "let x = 1"
            Expect.isSome (Map.tryFind "string_length" env) "builtin 'string_length' should be in env"
        }

        test "TE-02: all initialTypeEnv builtins present in env" {
            let env = typeCheckGetEnv "let x = 1"
            for kv in TypeCheck.initialTypeEnv do
                Expect.isSome
                    (Map.tryFind kv.Key env)
                    (sprintf "builtin '%s' should be in env" kv.Key)
        }

        // TE-01 + TE-02 combined: user + builtins coexist
        test "TE-01+02: user binding and builtins coexist in same env" {
            let env = typeCheckGetEnv "let myFunc x = x + 1"
            Expect.isSome (Map.tryFind "myFunc" env) "user binding 'myFunc' in env"
            Expect.isSome (Map.tryFind "print" env) "builtin 'print' in env"
        }

        // exportBindingEnv is identity (returns same map)
        test "exportBindingEnv returns same map" {
            let env = typeCheckGetEnv "let x = 1"
            let exported = TypeCheck.exportBindingEnv env
            Expect.equal exported env "exportBindingEnv should be identity"
        }

        // Queryability: Map.tryFind works for lookup (TE success criteria #3)
        test "env is queryable by name" {
            let env = typeCheckGetEnv "let foo = \"hello\""
            match Map.tryFind "foo" env with
            | Some (Type.Scheme([], [], Type.TString)) -> ()
            | Some other -> failwith (sprintf "Expected string scheme, got %A" other)
            | None -> failwith "foo not found"
        }
    ]
```
