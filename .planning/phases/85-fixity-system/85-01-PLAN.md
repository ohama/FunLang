**Data structure:**

```fsharp
type FixityInfo = { Assoc: Assoc; Prec: int }
type FixityEnv = Map<string, FixityInfo>
```

**Default fixity from first character** (mirrors Lexer.classifyOperator):

```fsharp
let defaultFixity (op: string) : FixityInfo =
    // Strip leading dots for classification (same as Lexer.classifyOperator)
    let eff = op.TrimStart('.')
    if eff.Length = 0 then { Assoc = Left; Prec = 7 }  // INFIXOP3 = multiplicative
    elif eff.Length >= 2 && eff.[0] = '*' && eff.[1] = '*' then { Assoc = Right; Prec = 8 } // INFIXOP4
    else
        match eff.[0] with
        | '=' | '<' | '>' | '|' | '&' | '$' | '!' -> { Assoc = Left; Prec = 4 }   // INFIXOP0
        | '@' | '^' -> { Assoc = Right; Prec = 5 }                                  // INFIXOP1
        | '+' | '-' -> { Assoc = Left; Prec = 6 }                                   // INFIXOP2
        | '*' | '/' | '%' -> { Assoc = Left; Prec = 7 }                             // INFIXOP3
        | _ -> { Assoc = Left; Prec = 4 }                                            // fallback
```

Precedence mapping: INFIXOP0=4, INFIXOP1=5, INFIXOP2=6, INFIXOP3=7, INFIXOP4=8.
These numbers are chosen so user-supplied `#[left N]` values can sit anywhere in the range.

**Collect fixity from declarations:**

```fsharp
let collectFixity (existing: FixityEnv) (decls: Decl list) : FixityEnv =
    (existing, decls) ||> List.fold (fun env decl ->
        match decl with
        | InfixDecl(attrs, name, _, _) ->
            match attrs |> List.tryPick (function FixityAttr(assoc, prec) -> Some(assoc, prec)) with
            | Some(assoc, prec) -> Map.add name { Assoc = assoc; Prec = prec } env
            | None -> env
        | _ -> env)
```

**Lookup with fallback:**

```fsharp
let lookupFixity (env: FixityEnv) (op: string) : FixityInfo =
    match Map.tryFind op env with
    | Some info -> info
    | None -> defaultFixity op
```

**Post-parse AST rewrite — rewriteFixity:**

Walk the AST. When encountering a left-leaning chain of infix applications
`App(App(Var(op2), App(App(Var(op1), a), b)), c)` (which represents `(a op1 b) op2 c`
as parsed by LALR), check if the operators' fixity in FixityEnv requires restructuring.

The key insight: The LALR parser groups ALL same-level INFIXOPn operators left-to-right
(except INFIXOP1 and INFIXOP4 which are right). When a user puts `#[right 6]` on an
operator that is INFIXOP2 (left, prec 6 by default), the tree needs to be restructured
from left-associative to right-associative.

Implement this as a two-step process:

1. **Flatten**: Given a nested `App(App(Var(op), lhs), rhs)` tree, collect into a flat
   list of `(operand, operator)` pairs — but only flatten operators at the SAME LALR
   precedence level (same INFIXOPn). Stop flattening when encountering a different
   operator precedence level.

2. **Rebuild**: Using the FixityEnv, rebuild the flat chain with correct associativity.
   For left-assoc: fold-left. For right-assoc: fold-right.
   If operators in a chain have different FixityEnv precedences, sort by precedence
   (higher binds tighter) and rebuild accordingly using a simple precedence-climbing
   algorithm.

```fsharp
/// Flatten a chain of infix applications that the LALR parser grouped together
/// (same INFIXOPn level). Returns (operands, operators) where
/// operands has one more element than operators.
let rec private flattenInfixChain (expr: Expr) : (Expr list * string list) option =
    match expr with
    | App(App(Var(op, _), lhs, _), rhs, _) when isOperator op ->
        match flattenInfixChain lhs with
        | Some(operands, ops) -> Some(operands @ [rhs], ops @ [op])
        | None -> Some([lhs; rhs], [op])
    | _ -> None

/// Check if a name looks like an operator (starts with op chars or is surrounded by parens in source)
let private isOperator (name: string) : bool =
    name.Length > 0 &&
    let c = name.[0]
    c = '!' || c = '#' || c = '$' || c = '%' || c = '&' || c = '*' ||
    c = '+' || c = '-' || c = '.' || c = '/' || c = ':' || c = '<' ||
    c = '=' || c = '>' || c = '?' || c = '@' || c = '\\' || c = '^' ||
    c = '|' || c = '~'

/// Rebuild a flat chain [a; b; c] with ops [op1; op2] using fixity info
let private rebuildChain (env: FixityEnv) (operands: Expr list) (ops: string list) (span: Span) : Expr =
    // Simple case: all operators are the same
    if ops |> List.distinct |> List.length = 1 then
        let info = lookupFixity env ops.[0]
        match info.Assoc with
        | Left ->
            // fold left: ((a op b) op c)
            let first = operands.[0]
            List.fold2 (fun acc op rhs ->
                App(App(Var(op, span), acc, span), rhs, span)
            ) first ops (List.tail operands)
        | Right ->
            // fold right: (a op (b op c))
            let revOps = List.rev ops
            let revOperands = List.rev operands
            let last = revOperands.[0]
            List.fold2 (fun acc op lhs ->
                App(App(Var(op, span), lhs, span), acc, span)
            ) last revOps (List.tail revOperands)
    else
        // Mixed operators: use precedence climbing
        // Sort by precedence and rebuild with tighter-binding ops first
        rebuildByPrecedence env operands ops span

/// Precedence climbing for mixed-operator chains
let private rebuildByPrecedence (env: FixityEnv) (operands: Expr list) (ops: string list) (span: Span) : Expr =
    // Convert to mutable array for in-place reduction
    let mutable items = operands |> List.toArray
    let mutable operators = ops |> List.toArray
    // Process from highest precedence to lowest
    let precGroups =
        operators
        |> Array.mapi (fun i op -> (i, lookupFixity env op))
        |> Array.sortByDescending (fun (_, info) -> info.Prec)
        |> Array.groupBy (fun (_, info) -> info.Prec)
    for (_, group) in precGroups do
        // Within same precedence, apply associativity
        let info = snd group.[0]
        let indices = group |> Array.map fst |> Array.sort
        match info.Assoc with
        | Left ->
            for i in indices do
                if i < operators.Length && operators.[i] <> "" then
                    // Reduce: items[i] op items[i+1] -> result at items[i], mark items[i+1] as consumed
                    let combined = App(App(Var(operators.[i], span), items.[i], span), items.[i+1], span)
                    items.[i] <- combined
                    // Shift remaining
                    ... (use a list-based approach instead)
        | Right -> ...
    // Return final item

    // ACTUALLY: For Phase 85, the simpler approach is better. The LALR parser already
    // handles mixed-precedence operators correctly because they're at different grammar
    // levels (INFIXOP0 vs INFIXOP2 etc). The only chains that get flattened are
    // same-INFIXOPn-level operators. So the "all operators same" case plus
    // "same-level but different FixityEnv prec" is what we need.
    //
    // For same-level different-FixityEnv-prec: use simple precedence climbing.
    // This is rare in practice and will mostly matter in Phase 86.

/// Main rewrite: walk entire AST, rewriting infix chains based on FixityEnv
let rewriteFixity (env: FixityEnv) (m: Module) : Module =
    // If env is empty, no rewriting needed (optimization for no-attribute case)
    if Map.isEmpty env then m
    else
        let rec rewriteExpr (e: Expr) : Expr =
            match flattenInfixChain e with
            | Some(operands, ops) when needsRewrite env ops ->
                let rewrittenOperands = operands |> List.map rewriteExpr
                rebuildChain env rewrittenOperands ops (Expr.span e)
            | _ ->
                // Recurse into sub-expressions
                mapExprChildren rewriteExpr e
        let rewriteDecl (d: Decl) : Decl =
            match d with
            | LetDecl(name, body, span) -> LetDecl(name, rewriteExpr body, span)
            | InfixDecl(attrs, name, body, span) -> InfixDecl(attrs, name, rewriteExpr body, span)
            | _ -> d  // other decls don't contain operator expressions that need rewriting
        let rewriteDecls = List.map rewriteDecl
        match m with
        | Module(decls, span) -> Module(rewriteDecls decls, span)
        | NamedModule(name, decls, span) -> NamedModule(name, rewriteDecls decls, span)
        | EmptyModule span -> EmptyModule span

/// Check if any operator in chain has fixity differing from LALR assignment
let private needsRewrite (env: FixityEnv) (ops: string list) : bool =
    ops |> List.exists (fun op ->
        match Map.tryFind op env with
        | Some info ->
            let dflt = defaultFixity op
            info.Assoc <> dflt.Assoc || info.Prec <> dflt.Prec
        | None -> false)
```

```
let (+++) x y = x + y + 1
```

`tests/flt/file/attribute/fixity-right-assoc-chain.flt`:
Test right-associative chaining with a power-like operator.
```
// Test: right-associative chained operator
// --- Command: src/FunLang/bin/Release/net10.0/fn %input
// --- Input:
#[right 8]
let (^^^) x y = to_int (to_float x ** to_float y)

// Right-assoc: 2 ^^^ (3 ^^^ 2) = 2 ^^^ 9 = 512
// Left-assoc would give: (2 ^^^ 3) ^^^ 2 = 8 ^^^ 2 = 64
let result = 2 ^^^ 3 ^^^ 2
let _ = println (to_string result)
// --- Output:
512
```

Note: `^^^` starts with `^` which is INFIXOP1 (right, prec 5). With `#[right 8]` it
stays right-assoc but at higher prec. Since it's already right-assoc, this tests that
the rewrite preserves correct right-associativity even when precedence changes.

Actually, `^` is INFIXOP1 which is already right-assoc. To properly test LEFT->RIGHT
override, use an operator starting with `+` (INFIXOP2, left):

```

// With left-assoc: (1 +++ 2) +++ 3 = 12 +++ 3 = 123
```
